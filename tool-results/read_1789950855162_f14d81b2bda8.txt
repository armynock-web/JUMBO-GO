   757→ts
   758→// app/api/driver/availability/online/route.ts
   759→
   760→import { NextResponse } from "next/server";
   761→import { createClient } from "@/lib/supabase/server";
   762→
   763→export async function POST() {
   764→  const supabase = await createClient();
   765→
   766→  const {
   767→    data: { user },
   768→  } = await supabase.auth.getUser();
   769→
   770→  if (!user) {
   771→    return NextResponse.json(
   772→      { error: "UNAUTHORIZED" },
   773→      { status: 401 }
   774→    );
   775→  }
   776→
   777→  const { data: driver } = await supabase
   778→    .from("driver_profiles")
   779→    .select("id, kyc_status, availability_status")
   780→    .eq("user_id", user.id)
   781→    .single();
   782→
   783→  if (!driver) {
   784→    return NextResponse.json(
   785→      { error: "DRIVER_NOT_FOUND" },
   786→      { status: 404 }
   787→    );
   788→  }
   789→
   790→  if (driver.kyc_status !== "APPROVED") {
   791→    return NextResponse.json(
   792→      {
   793→        error: "KYC_NOT_APPROVED",
   794→        message: "บัญชียังไม่ผ่านการตรวจสอบ",
   795→      },
   796→      { status: 422 }
   797→    );
   798→  }
   799→
   800→  const { data, error } = await supabase
   801→    .from("driver_profiles")
   802→    .update({
   803→      availability_status: "ONLINE",
   804→      updated_at: new Date().toISOString(),
   805→    })
   806→    .eq("id", driver.id)
   807→    .select()
   808→    .single();
   809→
   810→  if (error) {
   811→    return NextResponse.json(
   812→      { error: "UPDATE_FAILED" },
   813→      { status: 500 }
   814→    );
   815→  }
   816→
   817→  return NextResponse.json({
   818→    success: true,
   819→    driver: data,
   820→  });
   821→}
   822→10. การรับงานแบบปลอดภัย
   823→การรับงานต้องใช้ Transaction หรือ RPC เพื่อป้องกันคนขับสองคนกดรับงานเดียวกันพร้อมกัน

   825→sql
   826→create or replace function accept_job(
   827→  p_job_id uuid,
   828→  p_driver_id uuid
   829→)
   830→returns public.jobs
   831→language plpgsql
   832→security definer
   833→as $$
   834→declare
   835→  v_job public.jobs;
   836→begin
   837→  select *
   838→  into v_job
   839→  from public.jobs
   840→  where id = p_job_id
   841→  for update;
   842→
   843→  if v_job.id is null then
   844→    raise exception 'JOB_NOT_FOUND';
   845→  end if;
   846→
   847→  if v_job.status <> 'OFFERED' then
   848→    raise exception 'JOB_ALREADY_TAKEN';
   849→  end if;
   850→
   851→  update public.jobs
   852→  set
   853→    driver_id = p_driver_id,
   854→    status = 'ACCEPTED',
   855→    accepted_at = now(),
   856→    updated_at = now()
   857→  where id = p_job_id
   858→  returning * into v_job;
   859→
   860→  update public.driver_profiles
   861→  set availability_status = 'BUSY'
   862→  where id = p_driver_id;
   863→
   864→  return v_job;
   865→end;
   866→$$;
