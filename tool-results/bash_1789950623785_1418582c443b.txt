ts
// app/api/driver/availability/online/route.ts

import { NextResponse } from "next/server";
import { createClient } from "@/lib/supabase/server";

export async function POST() {
  const supabase = await createClient();

  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user) {
    return NextResponse.json(
      { error: "UNAUTHORIZED" },
      { status: 401 }
    );
  }

  const { data: driver } = await supabase
    .from("driver_profiles")
    .select("id, kyc_status, availability_status")
    .eq("user_id", user.id)
    .single();

  if (!driver) {
    return NextResponse.json(
      { error: "DRIVER_NOT_FOUND" },
      { status: 404 }
    );
  }

  if (driver.kyc_status !== "APPROVED") {
    return NextResponse.json(
      {
        error: "KYC_NOT_APPROVED",
        message: "บัญชียังไม่ผ่านการตรวจสอบ",
      },
      { status: 422 }
    );
  }

  const { data, error } = await supabase
    .from("driver_profiles")
    .update({
      availability_status: "ONLINE",
      updated_at: new Date().toISOString(),
    })
    .eq("id", driver.id)
    .select()
    .single();

  if (error) {
    return NextResponse.json(
      { error: "UPDATE_FAILED" },
      { status: 500 }
    );
  }

  return NextResponse.json({
    success: true,
    driver: data,
  });
}
10. การรับงานแบบปลอดภัย
การรับงานต้องใช้ Transaction หรือ RPC เพื่อป้องกันคนขับสองคนกดรับงานเดียวกันพร้อมกัน

sql
create or replace function accept_job(
  p_job_id uuid,
  p_driver_id uuid
)
returns public.jobs
language plpgsql
security definer
as $$
declare
  v_job public.jobs;
begin
  select *
  into v_job
  from public.jobs
  where id = p_job_id
  for update;

  if v_job.id is null then
    raise exception 'JOB_NOT_FOUND';
  end if;

  if v_job.status <> 'OFFERED' then
    raise exception 'JOB_ALREADY_TAKEN';
  end if;

  update public.jobs
  set
    driver_id = p_driver_id,
    status = 'ACCEPTED',
    accepted_at = now(),
    updated_at = now()
  where id = p_job_id
  returning * into v_job;

  update public.driver_profiles
  set availability_status = 'BUSY'
  where id = p_driver_id;

  return v_job;
end;
$$;
ห้ามให้ Client อัปเดต driver_id และ status โดยตรงเอง เพราะอาจทำให้รับงานซ้ำหรือปลอมสถานะได้
